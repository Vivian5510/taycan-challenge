set(TAYCAN_F2_MAC_SYNC "fullfsync" CACHE STRING "macOS file synchronization: fsync or fullfsync")

  message(FATAL_ERROR "Unknown macOS synchronization policy")
endif()
set(TAYCAN_F2_NATIVE_IO "inline" CACHE STRING "macOS diagnostic backend: inline or posix-aio")
set_property(CACHE TAYCAN_F2_NATIVE_IO PROPERTY STRINGS inline posix-aio)
option(TAYCAN_F2_NATIVE_IO_PROFILE "Count I/O executed by Get callers in native backend experiments" OFF)
if(NOT TAYCAN_F2_NATIVE_IO MATCHES "^(inline|posix-aio)$")
  message(FATAL_ERROR "Unknown native I/O backend")
endif()
if(NOT APPLE AND NOT TAYCAN_F2_NATIVE_IO STREQUAL "inline")
  message(FATAL_ERROR "Native POSIX AIO experiment is macOS only")
endif()
set(TAYCAN_F2_LOG_PAGE_BITS 20 CACHE STRING "F2 log page address bits")
set(TAYCAN_F2_OVERFLOW_PAGE_BITS 12 CACHE STRING "F2 overflow allocator page bits")
set(TAYCAN_F2_PREFIX_BLOCK_BYTES 4096 CACHE STRING "Native data-log seal unit (512 or 4096), explicit experimental format")
if(NOT TAYCAN_F2_PREFIX_BLOCK_BYTES MATCHES "^(512|4096)$")
  message(FATAL_ERROR "Unsupported native prefix alignment")
endif()
option(TAYCAN_F2_FIX_RECOVERY_VERSION "Preserve recovered version before clearing checkpoint metadata" ON)
option(TAYCAN_F2_FIX_RECOVERY_FRONTIER "Preserve committed flushed frontier when restoring the log" ON)
option(TAYCAN_F2_COMMIT_EXPERIMENT "Opt-in F2 internal commit and checkpoint wakeup experiment" OFF)
option(TAYCAN_F2_DURABLE_COMPACTION "Coordinate native F2 compaction with durable recovery roots" OFF)
set(TAYCAN_F2_DATA_SEGMENT_BITS 30 CACHE STRING "Data-log segment bits (new on-disk layout)")
set(TAYCAN_F2_SYNC_STRATEGY "native" CACHE STRING "native, data-log, hybridlog, checkpoint, or commit-log")
set_property(CACHE TAYCAN_F2_SYNC_STRATEGY PROPERTY STRINGS native data-log hybridlog checkpoint commit-log)
if(NOT TAYCAN_F2_SYNC_STRATEGY MATCHES "^(native|data-log|hybridlog|checkpoint|commit-log)$")
  message(FATAL_ERROR "Unknown F2 strategy")
endif()
set(F2_SYNC_EXPERIMENT 0)
set(F2_HYBRID_EXPERIMENT 0)
set(F2_PREFIX_EXPERIMENT 0)

    message(FATAL_ERROR "data-log experiment currently requires the audited macOS I/O tracking backend")
  endif()
  set(F2_PREFIX_EXPERIMENT 1)
endif()
if(TAYCAN_F2_SYNC_STRATEGY STREQUAL "hybridlog")
  set(F2_HYBRID_EXPERIMENT 1)
endif()

  set(TAYCAN_F2_COMMIT_EXPERIMENT ON)
  set(F2_SYNC_EXPERIMENT 1)
endif()

  set(F2_COMMIT_EXPERIMENT 1)
else()
  set(F2_COMMIT_EXPERIMENT 0)

  set(F2_RECOVERY_FIX 1)

  set(F2_FRONTIER_FIX 1)

  set(F2_FRONTIER_FIX 0)

  "${FASTER_SOURCE}" --log-page-bits "${TAYCAN_F2_LOG_PAGE_BITS}"
  --overflow-page-bits "${TAYCAN_F2_OVERFLOW_PAGE_BITS}"
  --fix-recovery-version "${F2_RECOVERY_FIX}" --fix-recovery-frontier "${F2_FRONTIER_FIX}"
  --commit-experiment "${F2_COMMIT_EXPERIMENT}"
  --sync-experiment "${F2_SYNC_EXPERIMENT}"
  --hybrid-experiment "${F2_HYBRID_EXPERIMENT}"
  --prefix-experiment "${F2_PREFIX_EXPERIMENT}"
  COMMAND_ERROR_IS_FATAL ANY)
if(APPLE)
  set(FASTER_FILE_SOURCE "${CMAKE_CURRENT_LIST_DIR}/native_file.cc")

  find_library(AIO_LIBRARY aio REQUIRED)
  find_library(UUID_LIBRARY uuid REQUIRED)
  set(FASTER_FILE_SOURCE "${FASTER_SOURCE}/environment/file_linux.cc")
