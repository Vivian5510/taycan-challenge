cmake_minimum_required(VERSION 3.24)
project(taycan_participant_sdk VERSION 1.0.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

find_package(Threads REQUIRED)

add_library(contest_sdk INTERFACE)
target_include_directories(contest_sdk INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(contest_sdk INTERFACE cxx_std_17)
target_link_libraries(contest_sdk INTERFACE Threads::Threads)
if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 8.3)
    message(FATAL_ERROR "GCC >= 8.3 is required; enable devtoolset-8 on CentOS 7")
  endif()
  if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
    target_link_libraries(contest_sdk INTERFACE stdc++fs)
  endif()
endif()

function(contest_add_executable target)
  add_executable(${target} ${ARGN})
  target_link_libraries(${target} PRIVATE contestant_engine contest_sdk)
endfunction()

# This research workspace develops F2; the original self-built implementation
# was retired. Keep the override for building the RocksDB adapter separately.
set(CONTEST_ENGINE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../faster-native-adapter"
    CACHE PATH "Engine source directory")
if(NOT EXISTS "${CONTEST_ENGINE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Engine source missing: ${CONTEST_ENGINE_DIR}. Use a fresh build directory or set CONTEST_ENGINE_DIR to the F2 adapter or RocksDB baseline.")
endif()
add_subdirectory("${CONTEST_ENGINE_DIR}" "${CMAKE_BINARY_DIR}/engine")

if(NOT TARGET contestant_engine)
  message(FATAL_ERROR "Submission must define contestant_engine")
endif()

get_target_property(engine_type contestant_engine TYPE)
if(NOT engine_type STREQUAL "STATIC_LIBRARY")
  message(FATAL_ERROR "contestant_engine must be a STATIC library")
endif()

# Tests alone depend on GoogleTest; engines and private benchmarks do not.
set(gtest_archive "${CMAKE_CURRENT_SOURCE_DIR}/third_party/googletest-1.14.0.tar.gz")
file(SHA256 "${gtest_archive}" gtest_digest)
if(NOT gtest_digest STREQUAL "8ad598c73ad796e0d8280b082cebd82a630d73e73cd3c70057938a6501bba5d7")
  message(FATAL_ERROR "Bundled GoogleTest source checksum mismatch")
endif()
set(gtest_source "${CMAKE_BINARY_DIR}/_deps/googletest-1.14.0")
if(NOT EXISTS "${gtest_source}/CMakeLists.txt")
  file(ARCHIVE_EXTRACT INPUT "${gtest_archive}" DESTINATION "${CMAKE_BINARY_DIR}/_deps")
endif()
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(gtest_build_tests OFF CACHE BOOL "" FORCE)
add_subdirectory("${gtest_source}" "${CMAKE_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)

# gflags is used only by the public test programs.
set(gflags_archive "${CMAKE_CURRENT_SOURCE_DIR}/third_party/gflags-2.2.2.tar.gz")
file(SHA256 "${gflags_archive}" gflags_digest)
if(NOT gflags_digest STREQUAL "34af2f15cf7367513b352bdcd2493ab14ce43692d2dcd9dfc499492966c64dcf")
  message(FATAL_ERROR "Bundled gflags source checksum mismatch")
endif()
set(gflags_source "${CMAKE_BINARY_DIR}/_deps/gflags-2.2.2")
if(NOT EXISTS "${gflags_source}/CMakeLists.txt")
  file(ARCHIVE_EXTRACT INPUT "${gflags_archive}" DESTINATION "${CMAKE_BINARY_DIR}/_deps")
endif()
set(GFLAGS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(GFLAGS_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GFLAGS_BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
function(build_test_flags)
  # gflags 2.2.2 predates CMake 4; limit this compatibility setting to it.
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
  add_subdirectory("${gflags_source}" "${CMAKE_BINARY_DIR}/gflags" EXCLUDE_FROM_ALL)
endfunction()
build_test_flags()
find_package(Python3 3.6 REQUIRED COMPONENTS Interpreter)

function(test_executable target)
  contest_add_executable(${target} ${ARGN} tests/test_options.cc)
  target_link_libraries(${target} PRIVATE gflags::gflags GTest::gtest)
endfunction()

foreach(name basic sequential random overwrite delete mixed recovery checkpoint
             concurrent demand_load memory_limit compaction)
  test_executable(${name}_test tests/${name}_test.cc tests/test_main.cc)
endforeach()
test_executable(crash_worker tests/crash_worker.cc)

    target_link_libraries(taycan_bench_fixture PRIVATE contest_sdk)
  endif()
