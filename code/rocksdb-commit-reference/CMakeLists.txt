cmake_minimum_required(VERSION 3.20)
project(rocksdb_commit_reference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(ROCKSDB_SOURCE "${ROOT}/research/rocksdb-v8.9.1" CACHE PATH "Pinned RocksDB source")
set(ROCKSDB_LIBRARY "${ROOT}/archive/build-history/deps/rocksdb-8.9.1-native-arm64/lib/librocksdb.a" CACHE FILEPATH "Matching RocksDB static library")
find_package(Threads REQUIRED)
add_executable(rocksdb_commit_probe probe.cc)
target_include_directories(rocksdb_commit_probe PRIVATE "${ROCKSDB_SOURCE}" "${ROCKSDB_SOURCE}/include")
target_compile_definitions(rocksdb_commit_probe PRIVATE ROCKSDB_PLATFORM_POSIX ROCKSDB_LIB_IO_POSIX)
if(APPLE)
  target_compile_definitions(rocksdb_commit_probe PRIVATE OS_MACOSX HAVE_FULLFSYNC)
endif()
target_compile_options(rocksdb_commit_probe PRIVATE -fno-rtti -Wall -Wextra -Wno-unused-parameter)
target_link_libraries(rocksdb_commit_probe PRIVATE "${ROCKSDB_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
